ヘルパースクリプト (任意・Windows標準のPowerShellのみ)

CORSは実測で問題なかったため、WoLを使いたい場合のみ必要です。常駐不要で、起動しなければ一切使われません。 保存して powershell -ExecutionPolicy Bypass -File bravia-helper.ps1 で起動し、 プロキシURLに http://127.0.0.1:8099/、WoLエンドポイントに http://127.0.0.1:8099/wol を設定します。

# bravia-helper.ps1 : CORS proxy + Wake-on-LAN  (PowerShell 5.1 / Windows standard)
$Prefix = 'http://127.0.0.1:8099/'
$listener = New-Object System.Net.HttpListener
$listener.Prefixes.Add($Prefix)
$listener.Start()
Write-Host "listening on $Prefix   (Ctrl+C to stop)"
while ($listener.IsListening) {
  $ctx = $listener.GetContext(); $req = $ctx.Request; $res = $ctx.Response
  $res.AddHeader('Access-Control-Allow-Origin','*')
  $res.AddHeader('Access-Control-Allow-Headers','*')
  $res.AddHeader('Access-Control-Allow-Methods','POST,GET,OPTIONS')
  if ($req.HttpMethod -eq 'OPTIONS') { $res.StatusCode = 204; $res.Close(); continue }
  $out = ''
  try {
    if ($req.Url.AbsolutePath -eq '/wol') {
      $mac = ($req.QueryString['mac'] -replace '[^0-9A-Fa-f]','')
      if ($mac.Length -ne 12) { throw 'bad mac' }
      $m = for ($i=0; $i -lt 12; $i+=2) { [byte]::Parse($mac.Substring($i,2),'HexNumber') }
      $pkt = [byte[]](@([byte]255)*6 + ($m*16))
      $udp = New-Object System.Net.Sockets.UdpClient
      $udp.EnableBroadcast = $true
      $ep = New-Object System.Net.IPEndPoint ([System.Net.IPAddress]::Broadcast), 9
      [void]$udp.Send($pkt, $pkt.Length, $ep); $udp.Close()
      $res.StatusCode = 200; $out = 'wol sent to ' + $mac
    } else {
      $target = $req.Url.PathAndQuery.TrimStart('/') -replace '^http:/(?!/)','http://'
      $body = (New-Object IO.StreamReader($req.InputStream, $req.ContentEncoding)).ReadToEnd()
      $w = [System.Net.HttpWebRequest]::Create($target)
      $w.Method = $req.HttpMethod
      $w.Timeout = 5000
      if ($req.ContentType) { $w.ContentType = $req.ContentType }
      foreach ($h in @('X-Auth-PSK','SOAPAction','X-CERS-DEVICE-ID')) {
        if ($req.Headers[$h]) { $w.Headers.Add($h, $req.Headers[$h]) }
      }
      if ($req.HttpMethod -eq 'POST') {
        $b = [Text.Encoding]::UTF8.GetBytes($body)
        $w.ContentLength = $b.Length
        $s = $w.GetRequestStream(); $s.Write($b,0,$b.Length); $s.Close()
      }
      $r = $w.GetResponse()
      $out = (New-Object IO.StreamReader($r.GetResponseStream())).ReadToEnd()
      $res.StatusCode = [int]$r.StatusCode
    }
  } catch [System.Net.WebException] {
    $er = $_.Exception.Response
    if ($er) { $res.StatusCode = [int]$er.StatusCode
               $out = (New-Object IO.StreamReader($er.GetResponseStream())).ReadToEnd() }
    else     { $res.StatusCode = 502; $out = $_.Exception.Message }
  } catch {
    $res.StatusCode = 500; $out = $_.Exception.Message
  }
  $buf = [Text.Encoding]::UTF8.GetBytes($out)
  $res.OutputStream.Write($buf,0,$buf.Length); $res.Close()
}
